require("dotenv").config();

const express = require("express");
const mongoose = require("mongoose");
const cors = require("cors");

// Routes
const authRoutes = require("./routes/authRoute");
const transactionRoutes = require("./routes/transactionRoute");
const categoryRoutes = require("./routes/categoryRoute");
const budgetRoutes = require("./routes/budgetRoute");
const reportRoutes = require("./routes/reportRoute");
const notificationRoutes = require("./routes/notificationRoute");
const recurringTransactionRoutes = require("./routes/recurringTransactionRoute");
const sharedExpenseRoutes = require("./routes/sharedExpenseRoutes");

const app = express();
const PORT = process.env.PORT || 5000;

app.disable("x-powered-by");

// Allowed frontend origins
const allowedOrigins = (
  process.env.CLIENT_URL || "http://localhost:5173"
)
  .split(",")
  .map((origin) => origin.trim().replace(/\/$/, ""))
  .filter(Boolean);

// Middleware
app.use(
  cors({
    origin: (origin, callback) => {
      // Allow requests without Origin (Postman/server-to-server).
      if (!origin || allowedOrigins.includes(origin.replace(/\/$/, ""))) {
        return callback(null, true);
      }

      return callback(new Error("Origin not allowed by CORS"));
    },
    credentials: true,
  })
);

app.use(express.json({ limit: "1mb" }));
app.use(express.urlencoded({ extended: true }));

// Health check
app.get("/", (req, res) => {
  res.status(200).json({
    success: true,
    message: "Finora Personal Finance API Running",
  });
});

// API routes
app.use("/api/auth", authRoutes);
app.use("/api/transactions", transactionRoutes);
app.use("/api/categories", categoryRoutes);
app.use("/api/budget", budgetRoutes);
app.use("/api/report", reportRoutes);
app.use("/api/notifications", notificationRoutes);
app.use("/api/recurring-transactions", recurringTransactionRoutes);
app.use("/api/shared-expenses", sharedExpenseRoutes);

// 404 handler
app.use((req, res) => {
  res.status(404).json({
    success: false,
    message: "API route not found",
  });
});

// Central error handler
app.use((err, req, res, next) => {
  console.error("API Error:", err.message);

  if (res.headersSent) {
    return next(err);
  }

  const status = err.status || err.statusCode || 500;

  res.status(status).json({
    success: false,
    message:
      status < 500
        ? err.message
        : "Internal server error.",
  });
});

// Start server after database connection
async function startServer() {
  try {
    if (!process.env.MONGO_URL) {
      throw new Error("MONGO_URL is missing in server .env");
    }

    await mongoose.connect(process.env.MONGO_URL);

    console.log("MongoDB Connected");

    app.listen(PORT, () => {
      console.log(`Finora server running on port ${PORT}`);
    });
  } catch (error) {
    console.error("Server startup failed:", error.message);
    process.exit(1);
  }
}

startServer();

"use strict";

const User = require("../models/user");
const bcrypt = require("bcryptjs");
const jwt = require("jsonwebtoken");
const crypto = require("crypto");
const nodemailer = require("nodemailer");

// Generate JWT
const generateToken = (user) => {
  if (!process.env.JWT_SECRET) {
    throw new Error("JWT_SECRET is missing in .env");
  }

  return jwt.sign(
    {
      id: user._id.toString(),
      email: user.email,
    },
    process.env.JWT_SECRET,
    { expiresIn: "7d" }
  );
};

// Safe user response
const sendUserResponse = (user) => ({
  id: user._id.toString(),
  _id: user._id.toString(),
  name: user.name,
  email: user.email,
  number: user.number || "",
});

// Register
const register = async (req, res) => {
  try {
    const { name, email, number, password } = req.body;

    if (
      typeof name !== "string" ||
      !name.trim() ||
      typeof email !== "string" ||
      !email.trim() ||
      typeof password !== "string" ||
      !password
    ) {
      return res.status(400).json({
        success: false,
        message: "Name, email and password are required.",
      });
    }

    if (password.length < 8) {
      return res.status(400).json({
        success: false,
        message: "Password must contain at least 8 characters.",
      });
    }

    const normalizedEmail = email.trim().toLowerCase();

    const existingUser = await User.findOne({
      email: normalizedEmail,
    });

    if (existingUser) {
      return res.status(409).json({
        success: false,
        message: "An account with this email already exists.",
      });
    }

    const hashedPassword = await bcrypt.hash(password, 12);

    const user = await User.create({
      name: name.trim(),
      email: normalizedEmail,
      number: typeof number === "string" ? number.trim() : "",
      password: hashedPassword,
    });

    return res.status(201).json({
      success: true,
      message: "Registration successful.",
      token: generateToken(user),
      user: sendUserResponse(user),
    });
  } catch (error) {
    console.error("Registration error:", error);

    return res.status(error.code === 11000 ? 409 : 500).json({
      success: false,
      message:
        error.code === 11000
          ? "An account with this email already exists."
          : "Registration failed.",
    });
  }
};

// Login
const login = async (req, res) => {
  try {
    const { email, password } = req.body;

    if (
      typeof email !== "string" ||
      !email.trim() ||
      typeof password !== "string" ||
      !password
    ) {
      return res.status(400).json({
        success: false,
        message: "Email and password are required.",
      });
    }

    // IMPORTANT: Explicitly select hidden password field.
    const user = await User.findOne({
      email: email.trim().toLowerCase(),
    }).select("+password");

    if (!user || !(await bcrypt.compare(password, user.password))) {
      return res.status(401).json({
        success: false,
        message: "Invalid email or password.",
      });
    }

    return res.status(200).json({
      success: true,
      message: "Login successful.",
      token: generateToken(user),
      user: sendUserResponse(user),
    });
  } catch (error) {
    console.error("Login error:", error);

    return res.status(500).json({
      success: false,
      message: "Login failed.",
    });
  }
};

// Forgot Password
const forgotPassword = async (req, res) => {
  try {
    const email =
      typeof req.body.email === "string"
        ? req.body.email.trim().toLowerCase()
        : "";

    if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
      return res.status(400).json({
        success: false,
        message: "Please enter a valid email address.",
      });
    }

    if (
      !process.env.EMAIL_USER ||
      !(process.env.EMAIL_PASS || process.env.EMAIL_PASSWORD) ||
      !process.env.CLIENT_URL
    ) {
      console.error("Password reset email configuration is incomplete.");

      return res.status(500).json({
        success: false,
        message: "Email service is not configured.",
      });
    }

    const message =
      "If an account matches that email, password reset instructions will be sent.";

    const user = await User.findOne({ email });

    if (!user) {
      return res.status(200).json({
        success: true,
        message,
      });
    }

    // Generate a secure random token.
    const resetToken = crypto.randomBytes(32).toString("hex");

    // Store only its SHA-256 hash.
    const hashedToken = crypto
      .createHash("sha256")
      .update(resetToken)
      .digest("hex");

    user.resetPasswordToken = hashedToken;
    user.resetPasswordExpires = new Date(
      Date.now() + 15 * 60 * 1000
    );

    await user.save();

    const clientUrl = process.env.CLIENT_URL.replace(/\/$/, "");

    const resetUrl =
      `${clientUrl}/reset-password?token=${resetToken}` +
      `&email=${encodeURIComponent(email)}`;

    const transporter = nodemailer.createTransport({
      service: "gmail",
      auth: {
        user: process.env.EMAIL_USER,
        pass: process.env.EMAIL_PASS || process.env.EMAIL_PASSWORD,
      },
    });

    try {
      await transporter.sendMail({
        from: `"Finora" <${process.env.EMAIL_USER}>`,
        to: email,
        subject: "Reset your Finora password",

        text: `Reset your password within 15 minutes using this link: ${resetUrl}`,

        html: `
          <div style="font-family:Arial,sans-serif;max-width:560px;margin:auto;padding:24px;color:#111827">
            <h2 style="color:#1729A8">Reset your password</h2>
            <p>We received a request to reset your Finora password.</p>
            <p>
              <a href="${resetUrl}"
                style="display:inline-block;background:#1729A8;color:#fff;padding:13px 22px;border-radius:8px;text-decoration:none">
                Reset Password
              </a>
            </p>
            <p>This link expires in 15 minutes and can only be used once.</p>
            <p>If you did not request this, ignore this email.</p>
          </div>
        `,
      });
    } catch (emailError) {
      user.resetPasswordToken = null;
      user.resetPasswordExpires = null;
      await user.save();
      throw emailError;
    }

    return res.status(200).json({
      success: true,
      message,
    });
  } catch (error) {
    console.error("Forgot password error:", error.message);

    return res.status(500).json({
      success: false,
      message: "Unable to send reset email. Please try again later.",
    });
  }
};

// Reset Password
const resetPassword = async (req, res) => {
  try {
    const { token, password } = req.body;

    if (
      typeof token !== "string" ||
      !token ||
      typeof password !== "string" ||
      !password
    ) {
      return res.status(400).json({
        success: false,
        message: "Reset token and new password are required.",
      });
    }

    if (password.length < 8) {
      return res.status(400).json({
        success: false,
        message: "Password must contain at least 8 characters.",
      });
    }

    // Hash the incoming token to match the database value.
    const hashedToken = crypto
      .createHash("sha256")
      .update(token)
      .digest("hex");

    // Only accept tokens that have not expired.
    const user = await User.findOne({
      resetPasswordToken: hashedToken,
      resetPasswordExpires: { $gt: new Date() },
    }).select("+resetPasswordToken +resetPasswordExpires");

    if (!user) {
      return res.status(400).json({
        success: false,
        message: "Reset link is invalid or has expired.",
      });
    }

    // Hash the new password.
    user.password = await bcrypt.hash(password, 12);

    // Invalidate the reset token after successful use.
    user.resetPasswordToken = null;
    user.resetPasswordExpires = null;

    await user.save();

    return res.status(200).json({
      success: true,
      message: "Password reset successful. Please log in.",
    });
  } catch (error) {
    console.error("Reset password error:", error);

    return res.status(500).json({
      success: false,
      message: "Unable to reset password. Please try again.",
    });
  }
};

module.exports = {
  register,
  login,
  forgotPassword,
  resetPassword,
};

const mongoose = require("mongoose");
const Budget = require("../models/Budget");

// GET /api/budget
const getBudget = async (req, res, next) => {
  try {
    const budgets = await Budget.find({
      user: req.user.id,
    }).sort({ createdAt: -1 });

    res.status(200).json({
      success: true,
      count: budgets.length,
      budgets,
    });
  } catch (error) {
    next(error);
  }
};

// GET /api/budget/:id
const getBudgetById = async (req, res, next) => {
  try {
    if (!mongoose.isValidObjectId(req.params.id)) {
      return res.status(400).json({
        success: false,
        message: "Invalid budget ID",
      });
    }

    const budget = await Budget.findOne({
      _id: req.params.id,
      user: req.user.id,
    });

    if (!budget) {
      return res.status(404).json({
        success: false,
        message: "Budget not found",
      });
    }

    res.status(200).json({ success: true, budget });
  } catch (error) {
    next(error);
  }
};

// POST /api/budget
const createBudget = async (req, res, next) => {
  try {
    const { category, limit, month, year } = req.body;

    if (!category || limit === undefined || limit === null) {
      return res.status(400).json({
        success: false,
        message: "Category and budget limit are required",
      });
    }

    const numericLimit = Number(limit);

    if (!Number.isFinite(numericLimit) || numericLimit <= 0) {
      return res.status(400).json({
        success: false,
        message: "Budget limit must be greater than zero",
      });
    }

    const budget = await Budget.create({
      user: req.user.id,
      category: String(category).trim(),
      limit: numericLimit,
      month: month ?? new Date().getMonth() + 1,
      year: year ?? new Date().getFullYear(),
    });

    res.status(201).json({
      success: true,
      message: "Budget created successfully",
      budget,
    });
  } catch (error) {
    next(error);
  }
};

// GET /api/budget/progress
const getBudgetProgress = async (req, res, next) => {
  try {
    const budgets = await Budget.find({
      user: req.user.id,
    }).sort({ createdAt: -1 });

    res.status(200).json({
      success: true,
      count: budgets.length,
      budgets,
    });
  } catch (error) {
    next(error);
  }
};

// PUT /api/budget/:id
const updateBudget = async (req, res, next) => {
  try {
    if (!mongoose.isValidObjectId(req.params.id)) {
      return res.status(400).json({
        success: false,
        message: "Invalid budget ID",
      });
    }

    const updates = {};
    const { category, limit, month, year } = req.body;

    if (category !== undefined) {
      if (!String(category).trim()) {
        return res.status(400).json({
          success: false,
          message: "Category cannot be empty",
        });
      }
      updates.category = String(category).trim();
    }

    if (limit !== undefined) {
      const numericLimit = Number(limit);

      if (!Number.isFinite(numericLimit) || numericLimit <= 0) {
        return res.status(400).json({
          success: false,
          message: "Budget limit must be greater than zero",
        });
      }
      updates.limit = numericLimit;
    }

    if (month !== undefined) {
      const numericMonth = Number(month);

      if (
        !Number.isInteger(numericMonth) ||
        numericMonth < 1 ||
        numericMonth > 12
      ) {
        return res.status(400).json({
          success: false,
          message: "Month must be between 1 and 12",
        });
      }
      updates.month = numericMonth;
    }

    if (year !== undefined) {
      const numericYear = Number(year);

      if (!Number.isInteger(numericYear) || numericYear < 2000) {
        return res.status(400).json({
          success: false,
          message: "Invalid budget year",
        });
      }
      updates.year = numericYear;
    }

    const budget = await Budget.findOneAndUpdate(
      { _id: req.params.id, user: req.user.id },
      { $set: updates },
      { new: true, runValidators: true }
    );

    if (!budget) {
      return res.status(404).json({
        success: false,
        message: "Budget not found",
      });
    }

    res.status(200).json({
      success: true,
      message: "Budget updated successfully",
      budget,
    });
  } catch (error) {
    next(error);
  }
};

// DELETE /api/budget/:id
const deleteBudget = async (req, res, next) => {
  try {
    if (!mongoose.isValidObjectId(req.params.id)) {
      return res.status(400).json({
        success: false,
        message: "Invalid budget ID",
      });
    }

    const budget = await Budget.findOneAndDelete({
      _id: req.params.id,
      user: req.user.id,
    });

    if (!budget) {
      return res.status(404).json({
        success: false,
        message: "Budget not found",
      });
    }

    res.status(200).json({
      success: true,
      message: "Budget deleted successfully",
    });
  } catch (error) {
    next(error);
  }
};

module.exports = {
  createBudget,
  getBudget,
  getBudgetById,
  updateBudget,
  deleteBudget,
  getBudgetProgress,
};

exports.getCategories = async (req, res, next) => {

function getNextDueDate(date, frequency) {
  const next = new Date(date);

  if (frequency === "weekly") {
    next.setDate(next.getDate() + 7);
  } else if (frequency === "monthly") {
    const originalDay = next.getDate();

    next.setDate(1);
    next.setMonth(next.getMonth() + 1);

    const lastDay = new Date(
      next.getFullYear(),
      next.getMonth() + 1,
      0
    ).getDate();

    next.setDate(Math.min(originalDay, lastDay));
  } else if (frequency === "yearly") {
    const originalMonth = next.getMonth();
    const originalDay = next.getDate();

    next.setDate(1);
    next.setFullYear(next.getFullYear() + 1);
    next.setMonth(originalMonth);

    const lastDay = new Date(
      next.getFullYear(),
      originalMonth + 1,
      0
    ).getDate();

    next.setDate(Math.min(originalDay, lastDay));
  }

  return next;
}

exports.updateRecurringTransaction = async (req, res, next) => {
  try {
    const userId = getUserId(req);

    if (!userId) {
      return sendError(res, 401, "Please log in again.");
    }

    const { id } = req.params;

    if (!mongoose.isValidObjectId(id)) {
      return sendError(res, 400, "Invalid recurring transaction ID.");
    }

    const allowedFields = [
      "title",
      "amount",
      "type",
      "category",
      "frequency",
      "nextDueDate",
      "endDate",
      "note",
      "isActive",
    ];

    const updates = {};

    for (const field of allowedFields) {
      if (req.body[field] !== undefined) {
        updates[field] = req.body[field];
      }
    }

    if (Object.keys(updates).length === 0) {
      return sendError(res, 400, "No valid fields provided to update.");
    }

    if (updates.title !== undefined) {
      if (typeof updates.title !== "string" || !updates.title.trim()) {
        return sendError(res, 400, "Title cannot be empty.");
      }

      updates.title = updates.title.trim();
    }

    if (updates.category !== undefined) {
      if (
        typeof updates.category !== "string" ||
        !updates.category.trim()
      ) {
        return sendError(res, 400, "Category cannot be empty.");
      }

      updates.category = updates.category.trim();
    }

    if (updates.note !== undefined) {
      if (typeof updates.note !== "string") {
        return sendError(res, 400, "Note must be text.");
      }

      updates.note = updates.note.trim();
    }

    if (updates.amount !== undefined) {
      if (
        updates.amount === "" ||
        updates.amount === null ||
        !Number.isFinite(Number(updates.amount)) ||
        Number(updates.amount) <= 0
      ) {
        return sendError(res, 400, "Amount must be greater than zero.");
      }

      updates.amount = Number(updates.amount);
    }

    if (
      updates.frequency !== undefined &&
      !validFrequencies.includes(updates.frequency)
    ) {
      return sendError(res, 400, "Invalid frequency.");
    }

    if (
      updates.type !== undefined &&
      !validTypes.includes(updates.type)
    ) {
      return sendError(res, 400, "Invalid transaction type.");
    }

    for (const field of ["nextDueDate", "endDate"]) {
      if (updates[field] === undefined) continue;

      if (updates[field] === null || updates[field] === "") {
        if (field === "endDate") {
          updates[field] = null;
          continue;
        }

        return sendError(res, 400, "Next due date is required.");
      }

      const parsedDate = new Date(updates[field]);

      if (Number.isNaN(parsedDate.getTime())) {
        return sendError(res, 400, `Invalid ${field}.`);
      }

      updates[field] = parsedDate;
    }

    if (
      updates.isActive !== undefined &&
      typeof updates.isActive !== "boolean"
    ) {
      return sendError(res, 400, "isActive must be true or false.");
    }

    const existing = await RecurringTransaction.findOne({
      _id: id,
      user: userId,
    });

    if (!existing) {
      return sendError(res, 404, "Recurring transaction not found.");
    }

    const resultingNextDueDate =
      updates.nextDueDate || existing.nextDueDate;

    const resultingEndDate =
      updates.endDate !== undefined
        ? updates.endDate
        : existing.endDate;

    if (
      resultingEndDate &&
      resultingNextDueDate > resultingEndDate
    ) {
      return sendError(
        res,
        400,
        "Next due date cannot be after the end date."
      );
    }

    const record = await RecurringTransaction.findOneAndUpdate(
      { _id: id, user: userId },
      { $set: updates },
      { new: true, runValidators: true }
    );

    return res.status(200).json({
      success: true,
      message: "Recurring transaction updated successfully.",
      record,
    });

exports.markRecurringPaid = async (req, res, next) => {
  try {
    const userId = getUserId(req);

    if (!userId) {
      return sendError(res, 401, "Please log in again.");
    }

    const { id } = req.params;

    if (!mongoose.isValidObjectId(id)) {
      return sendError(res, 400, "Invalid recurring transaction ID.");
    }

    const record = await RecurringTransaction.findOne({
      _id: id,
      user: userId,
    });

    if (!record) {
      return sendError(res, 404, "Recurring transaction not found.");
    }

    if (!record.isActive) {
      return sendError(res, 400, "This recurring schedule is paused.");
    }

    const dueDate = new Date(record.nextDueDate);

    if (Number.isNaN(dueDate.getTime())) {
      return sendError(res, 400, "The recurring schedule has an invalid due date.");
    }

    if (record.endDate && dueDate > new Date(record.endDate)) {
      await RecurringTransaction.updateOne(
        { _id: record._id, user: userId, nextDueDate: record.nextDueDate },
        { $set: { isActive: false } }
      );

      return sendError(res, 400, "This recurring schedule has ended.");
    }

    const occurrenceKey =
      `${record._id}:${dueDate.toISOString().slice(0, 10)}`;

    let transaction;

    try {
      transaction = await Transaction.findOneAndUpdate(
        { user: userId, occurrenceKey },
        {
          $setOnInsert: {
            user: userId,
            title: record.title,
            amount: record.amount,
            type: record.type,
            category: record.category,
            date: dueDate,
            note: record.note || "",
            recurringId: record._id,
            occurrenceKey,
          },
        },
        { new: true, upsert: true, runValidators: true }
      );
    } catch (error) {
      if (error.code === 11000) {
        transaction = await Transaction.findOne({
          user: userId,
          occurrenceKey,
        });
      }

      if (!transaction) throw error;
    }

    const nextDueDate = getNextDueDate(dueDate, record.frequency);

    const updates = {
      lastGeneratedDate: dueDate,
      nextDueDate,
    };

    if (record.endDate && nextDueDate > new Date(record.endDate)) {
      updates.isActive = false;
    }

    const updatedRecord = await RecurringTransaction.findOneAndUpdate(
      {
        _id: record._id,
        user: userId,
        nextDueDate: record.nextDueDate,
        isActive: true,
      },
      { $set: updates },
      { new: true, runValidators: true }
    );

      return res.status(409).json({
        success: false,
        message:
          "This payment was already processed or the schedule changed. Refresh and check the latest status.",

      });
    }

      message: "Payment recorded and next due date updated.",

const mongoose = require("mongoose");
const Transaction = require("../models/Transaction");

const VALID_PERIODS = [
  "thisMonth",
  "lastMonth",
  "last3Months",
  "thisYear",
  "allTime",
];

// Convert an IST calendar date to its UTC Date equivalent.
function istDateToUTC(year, month, day) {
  return new Date(Date.UTC(year, month - 1, day) - 330 * 60 * 1000);
}

// Get today's calendar date in India.
function getISTToday() {
  const parts = new Intl.DateTimeFormat("en-IN", {
    timeZone: "Asia/Kolkata",
    year: "numeric",
    month: "numeric",
    day: "numeric",
  }).formatToParts(new Date());

  const values = {};

  parts.forEach((part) => {
    if (part.type !== "literal") {
      values[part.type] = Number(part.value);
    }
  });

  return {
    year: values.year,
    month: values.month,
    day: values.day,
  };
}

// Build a date filter for the selected reporting period.
function getPeriodDateFilter(period) {
  if (!VALID_PERIODS.includes(period)) {
    return { error: true };
  }

  if (period === "allTime") {
    return { filter: null };
  }

  const { year, month } = getISTToday();

  let start;
  let end;

  switch (period) {
    case "thisMonth":
      start = istDateToUTC(year, month, 1);
      end =
        month === 12
          ? istDateToUTC(year + 1, 1, 1)
          : istDateToUTC(year, month + 1, 1);
      break;

    case "lastMonth": {
      const previousMonth = new Date(
        Date.UTC(year, month - 2, 1)
      );

      const previousYear = previousMonth.getUTCFullYear();
      const previousMonthNumber =
        previousMonth.getUTCMonth() + 1;

      start = istDateToUTC(
        previousYear,
        previousMonthNumber,
        1
      );
      end = istDateToUTC(year, month, 1);
      break;
    }

    case "last3Months": {
      const threeMonthStart = new Date(
        Date.UTC(year, month - 3, 1)
      );

      start = istDateToUTC(
        threeMonthStart.getUTCFullYear(),
        threeMonthStart.getUTCMonth() + 1,
        1
      );

      end =
        month === 12
          ? istDateToUTC(year + 1, 1, 1)
          : istDateToUTC(year, month + 1, 1);
      break;
    }

    case "thisYear":
      start = istDateToUTC(year, 1, 1);
      end = istDateToUTC(year + 1, 1, 1);
      break;

    default:
      return { error: true };
  }

  return {
    filter: {
      $gte: start,
      $lt: end,
    },
  };
}

function getUserId(req, res) {
  const userId = req.user?.id || req.user?._id;

  if (!userId || !mongoose.isValidObjectId(userId)) {
    res.status(401).json({
      success: false,
      message: "Authentication required. Please log in again.",
    });

    return null;
  }

  return new mongoose.Types.ObjectId(userId);
}

function getReportFilter(req, res) {
  const period = req.query.period || "allTime";
  const result = getPeriodDateFilter(period);

  if (result.error) {
    res.status(400).json({
      success: false,
      message: "Invalid report period.",
      allowedPeriods: VALID_PERIODS,
    });

    return null;
  }

  return result.filter;
}

// GET /api/report
const getDashboardReport = async (req, res, next) => {
  try {
    const userId = getUserId(req, res);
    if (!userId) return;

    const dateFilter = getReportFilter(req, res);
    if (dateFilter === null && req.query.period &&
        req.query.period !== "allTime") return;

    const match = { user: userId };

    if (dateFilter) {
      match.date = dateFilter;
    }

    const [result] = await Transaction.aggregate([
      { $match: match },
      {
        $group: {
          _id: null,
          totalIncome: {
            $sum: {
              $cond: [
                { $eq: ["$type", "income"] },
                "$amount",
                0,
              ],
            },
          },
          totalExpense: {
            $sum: {
              $cond: [
                { $eq: ["$type", "expense"] },
                "$amount",
                0,
              ],
            },
          },
          transactionCount: { $sum: 1 },
        },
      },
    ]);

    const totalIncome = result?.totalIncome || 0;
    const totalExpense = result?.totalExpense || 0;

    return res.status(200).json({
      success: true,
      period: req.query.period || "allTime",
      report: {
        totalIncome,
        totalExpense,
        balance: totalIncome - totalExpense,
        transactionCount: result?.transactionCount || 0,
      },
    });
  } catch (error) {

exports.createSharedExpense = async (req, res, next) => {
  try {
    const userId = requireUser(req, res);
    if (!userId) return;

    const {
      groupId,
      title,
      totalAmount,
      paidBy,
      category,
      date,
      splitType,
      splits,
      note,
    } = req.body;

    const amount = Number(totalAmount);

    if (!mongoose.isValidObjectId(groupId)) {
      return res.status(400).json({
        success: false,
        message: "Valid groupId is required.",
      });
    }

    if (
      typeof title !== "string" ||
      !title.trim() ||
      totalAmount === "" ||
      totalAmount === null ||
      !Number.isFinite(amount) ||
      amount <= 0
    ) {
      return res.status(400).json({
        success: false,
        message: "Title and a positive total amount are required.",
      });
    }

    if (!["equal", "custom", "percentage"].includes(splitType)) {
      return res.status(400).json({
        success: false,
        message: "Invalid split type.",
      });
    }

    let expenseDate = new Date();

    if (date !== undefined && date !== null && date !== "") {
      expenseDate = new Date(date);

      if (Number.isNaN(expenseDate.getTime())) {
        return res.status(400).json({
          success: false,
          message: "Invalid expense date.",
        });
      }
    }

    const group = await getMemberGroup(groupId, userId);

    if (!group) {
      return res.status(404).json({
        success: false,
        message: "Group not found.",
      });
    }

    const payerId = String(paidBy || userId);
    const memberIds = group.members.map((member) => String(member));

    if (!memberIds.includes(payerId)) {
      return res.status(400).json({
        success: false,
        message: "Payer must belong to this group.",
      });
    }

    let calculatedSplits = [];

    if (splitType === "equal") {
      const baseCents = Math.floor(
        Math.round(amount * 100) / memberIds.length
      );

      let remainingCents =
        Math.round(amount * 100) - baseCents * memberIds.length;

      calculatedSplits = memberIds.map((memberId) => {
        const extraCent = remainingCents > 0 ? 1 : 0;
        remainingCents -= extraCent;

        return {
          user: memberId,
          amount: (baseCents + extraCent) / 100,
          percentage: null,
        };
      });
    } else {
      if (
        !Array.isArray(splits) ||
        splits.length !== memberIds.length
      ) {
        return res.status(400).json({
          success: false,
          message: "Provide a split for every group member.",
        });
      }

      const suppliedIds = splits.map((split) => String(split.user));
      const uniqueIds = new Set(suppliedIds);

      if (
        uniqueIds.size !== memberIds.length ||
        !memberIds.every((id) => uniqueIds.has(id))
      ) {
        return res.status(400).json({
          success: false,
          message: "Each group member must appear exactly once in splits.",
        });
      }

      if (splitType === "custom") {
        const invalidAmount = splits.some(
          (split) =>
            split.amount === "" ||
            split.amount === null ||
            split.amount === undefined ||
            !Number.isFinite(Number(split.amount)) ||
            Number(split.amount) < 0
        );

        if (invalidAmount) {
          return res.status(400).json({
            success: false,
            message: "Invalid custom split amount.",
          });
        }

        calculatedSplits = splits.map((split) => ({
          user: String(split.user),
          amount: roundMoney(Number(split.amount)),
          percentage: null,
        }));

        const splitTotal = roundMoney(
          calculatedSplits.reduce(
            (sum, split) => sum + split.amount,
            0
          )
        );

        if (splitTotal !== roundMoney(amount)) {
          return res.status(400).json({
            success: false,
            message: "Custom split amounts must equal the total.",
            splitTotal,
            totalAmount: roundMoney(amount),
          });
        }
      } else {
        const invalidPercentage = splits.some((split) => {
          if (
            split.percentage === "" ||
            split.percentage === null ||
            split.percentage === undefined
          ) {
            return true;
          }

          const percentage = Number(split.percentage);

          return (
            !Number.isFinite(percentage) ||
            percentage < 0 ||
            percentage > 100
          );
        });

        if (invalidPercentage) {
          return res.status(400).json({
            success: false,
            message: "Each percentage must be between 0 and 100.",
          });
        }

        const percentageTotal = roundMoney(
          splits.reduce(
            (sum, split) => sum + Number(split.percentage),
            0
          )
        );

        if (percentageTotal !== 100) {
          return res.status(400).json({
            success: false,
            message: "Percentages must add up to exactly 100%.",
            percentageTotal,
          });
        }

        calculatedSplits = splits.map((split) => ({
          user: String(split.user),
          percentage: Number(split.percentage),
          amount: roundMoney(
            (amount * Number(split.percentage)) / 100
          ),
        }));

        const difference = roundMoney(
          amount -
            calculatedSplits.reduce(
              (sum, split) => sum + split.amount,
              0
            )
        );

        calculatedSplits[calculatedSplits.length - 1].amount =
          roundMoney(
            calculatedSplits[calculatedSplits.length - 1].amount +
              difference
          );
      }
    }

    const expense = await SharedExpense.create({
      group: group._id,
      createdBy: userId,
      paidBy: payerId,
      title: title.trim(),
      totalAmount: roundMoney(amount),
      category: category || "Other",
      date: expenseDate,
      splitType,
      splits: calculatedSplits,
      note: typeof note === "string" ? note.trim() : "",
    });

    const populated = await SharedExpense.findById(expense._id)
      .populate("paidBy", "name email")
      .populate("createdBy", "name email")
      .populate("splits.user", "name email");

    return res.status(201).json({
      success: true,
      expense: populated,
    });
  } catch (error) {
    next(error);
  }
};

const addTransaction = async (req, res, next) => {
  try {
    const userId = getUserId(req, res);
    if (!userId) return;

    const { title, amount, type, category, date, note } = req.body;

    if (
      typeof title !== "string" ||
      !title.trim() ||
      amount === undefined ||
      amount === null ||
      amount === "" ||
      !["income", "expense"].includes(type) ||
      typeof category !== "string" ||
      !category.trim()
    ) {
      return res.status(400).json({
        success: false,
        message: "Valid title, amount, type and category are required.",
      });
    }

    const numericAmount = Number(amount);

    if (!Number.isFinite(numericAmount) || numericAmount < 0) {
      return res.status(400).json({
        success: false,
        message: "Amount must be a valid non-negative number.",
      });
    }

    const transactionDate = date ? new Date(date) : new Date();

    if (Number.isNaN(transactionDate.getTime())) {
      return res.status(400).json({
        success: false,
        message: "Invalid transaction date.",
      });
    }

    const transaction = await Transaction.create({
      title: title.trim(),
      amount: numericAmount,
      type,
      category: category.trim(),
      date: transactionDate,
      note: typeof note === "string" ? note.trim() : "",
      user: userId,
    });

    return res.status(201).json({
      success: true,
      message: "Transaction created successfully.",
      transaction,
    });
  } catch (error) {
    next(error);
  }
};

const updateTransaction = async (req, res, next) => {
  try {
    const userId = getUserId(req, res);
    if (!userId) return;

    if (!mongoose.isValidObjectId(req.params.id)) {
      return res.status(400).json({
        success: false,
        message: "Invalid transaction ID.",
      });
    }

    const allowedFields = [
      "title",
      "amount",
      "type",
      "category",
      "date",
      "note",
    ];

    const updates = {};

    for (const field of allowedFields) {
      if (req.body[field] !== undefined) {
        updates[field] = req.body[field];
      }
    }

    if (updates.title !== undefined) {
      if (typeof updates.title !== "string" || !updates.title.trim()) {
        return res.status(400).json({
          success: false,
          message: "Title cannot be empty.",
        });
      }

      updates.title = updates.title.trim();
    }

    if (updates.amount !== undefined) {
      if (updates.amount === "" || updates.amount === null) {
        return res.status(400).json({
          success: false,
          message: "Amount is required.",
        });
      }

      updates.amount = Number(updates.amount);

      if (!Number.isFinite(updates.amount) || updates.amount < 0) {
        return res.status(400).json({
          success: false,
          message: "Amount must be a valid non-negative number.",
        });
      }
    }

    if (
      updates.type !== undefined &&
      !["income", "expense"].includes(updates.type)
    ) {
      return res.status(400).json({
        success: false,
        message: "Invalid transaction type.",
      });
    }

    if (updates.category !== undefined) {
      if (
        typeof updates.category !== "string" ||
        !updates.category.trim()
      ) {
        return res.status(400).json({
          success: false,
          message: "Category cannot be empty.",
        });
      }

      updates.category = updates.category.trim();
    }

    if (updates.date !== undefined) {
      const parsedDate = new Date(updates.date);

      if (Number.isNaN(parsedDate.getTime())) {
        return res.status(400).json({
          success: false,
          message: "Invalid transaction date.",
        });
      }

      updates.date = parsedDate;
    }

    if (updates.note !== undefined && typeof updates.note !== "string") {
      return res.status(400).json({
        success: false,
        message: "Note must be text.",
      });
    }

    if (updates.note !== undefined) {
      updates.note = updates.note.trim();
    }

    const transaction = await Transaction.findOneAndUpdate(
      {
        _id: req.params.id,
        user: userId,
      },
      { $set: updates },
      { new: true, runValidators: true }
    );

    if (!transaction) {
      return res.status(404).json({
        success: false,
        message: "Transaction not found.",
      });
    }

    return res.status(200).json({
      success: true,
      message: "Transaction updated successfully.",
      transaction,
    });
  } catch (error) {
    next(error);
  }
};

const jwt = require("jsonwebtoken");

const authMiddleware = (req, res, next) => {
  try {
    const authHeader = req.headers.authorization;

    if (!authHeader || !authHeader.startsWith("Bearer ")) {
      return res.status(401).json({
        success: false,
        message: "Authentication required. Please log in.",
      });
    }

    const token = authHeader.split(" ")[1];

    if (!process.env.JWT_SECRET) {
      return res.status(500).json({
        success: false,
        message: "JWT_SECRET is missing in server .env",
      });
    }

    const decoded = jwt.verify(token, process.env.JWT_SECRET);

    req.user = {
      id: decoded.id,
      email: decoded.email,
    };

    next();
  } catch (error) {
    return res.status(401).json({
      success: false,
      message:
        error.name === "TokenExpiredError"
          ? "Session expired. Please log in again."
          : "Invalid authentication token.",
    });
  }
};

module.exports = authMiddleware;

const errorHandler = (err, req, res, next) => {
  console.error("Error:", err.message);

  if (res.headersSent) {
    return next(err);
  }

  let statusCode = err.statusCode || err.status || 500;
  let message = err.message || "Internal server error";

  // Mongoose validation error
  if (err.name === "ValidationError") {
    statusCode = 400;
    message = Object.values(err.errors)
      .map((item) => item.message)
      .join(", ");
  }

  // Invalid MongoDB ID
  if (err.name === "CastError") {
    statusCode = 400;
    message = "Invalid ID or data format";
  }

  // Duplicate record
  if (err.code === 11000) {
    statusCode = 409;
    message = "This record already exists";
  }

  // Hide internal server details from clients
  if (statusCode >= 500) {
    message = "Internal server error";
  }

  res.status(statusCode).json({
    success: false,
    message,
  });
};

module.exports = errorHandler;

const validate = (schema) => {
  return (req, res, next) => {
    try {
      if (!schema || typeof schema.validate !== "function") {
        return res.status(500).json({
          success: false,
          message: "Invalid validation schema configuration",
        });
      }

      const result = schema.validate(req.body, {
        abortEarly: false,
        stripUnknown: true,
      });

      if (result.error) {
        return res.status(400).json({
          success: false,
          message: "Validation failed",
          errors: result.error.details.map(
            (detail) => detail.message
          ),
        });
      }

      req.body = result.value;
      return next();
    } catch (error) {
      return next(error);
    }
  };
};

module.exports.exportCSV = async (req, res, next) => {
  try {
    const userId = req.user?.id;

    if (!userId) {
      return res.status(401).json({
        success: false,
        message: "Authentication required.",
      });
    }

    const transactions = await Transaction.find({ user: userId })
      .select("title amount type category date note")
      .sort({ date: -1 })
      .lean();

    const fields = [
      { label: "Title", value: "title" },
      { label: "Amount", value: "amount" },
      { label: "Type", value: "type" },
      { label: "Category", value: "category" },
      { label: "Date", value: "date" },
      { label: "Note", value: "note" },
    ];

    const parser = new Parser({ fields });
    const csv = parser.parse(transactions);

    res.setHeader("Content-Type", "text/csv; charset=utf-8");
    res.setHeader(
      "Content-Disposition",
      'attachment; filename="transactions.csv"'
    );

    return res.status(200).send("\uFEFF" + csv);
  } catch (error) {
    return next(error);
  }
};
